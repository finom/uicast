import { useCallback, useEffect, useState } from "react";
import { Button } from "@/components/ui/button";
import { Card, CardContent } from "@/components/ui/card";

type Product = { id: number; name: string; price: number; qty: number };

const listProducts = (): Promise<Product[]> =>
  fetch("/api/products").then((res) => res.json());

const updateProduct = (id: number, patch: Partial<Product>) =>
  fetch(`/api/products/${id}`, { method: "PATCH", body: JSON.stringify(patch) });

export function Products() {
  const [products, setProducts] = useState<Product[]>([]);
  const [editingId, setEditingId] = useState<number | null>(null);
  const [draftName, setDraftName] = useState("");
  const [draftPrice, setDraftPrice] = useState(0);
  const refresh = useCallback(() => listProducts().then(setProducts), []);

  useEffect(() => {
    refresh();
  }, [refresh]);

  const setQty = (id: number, qty: number) => {
    setProducts((prev) => prev.map((p) => (p.id === id ? { ...p, qty } : p)));
    updateProduct(id, { qty });
  };

  const save = () => {
    if (!window.confirm("Apply these changes to the product?")) return;
    setProducts((prev) =>
      prev.map((p) =>
        p.id === editingId ? { ...p, name: draftName, price: draftPrice } : p,
      ),
    );
    if (editingId !== null)
      updateProduct(editingId, { name: draftName, price: draftPrice });
    setEditingId(null);
  };

  return (
    <Card className="w-72">
      <CardContent className="grid gap-2">
        <div className="font-semibold">{products.length} products</div>
        <Button variant="outline" size="sm" className="justify-self-start" onClick={refresh}>
          Refresh
        </Button>
        {products.map((product) => (
          <div key={product.id} className="flex items-center gap-2 border-t pt-1 text-sm">
            <span className="truncate">{product.name}</span>
            <Button
              variant="ghost"
              size="sm"
              className="mr-auto h-6 px-1.5"
              onClick={() => {
                setDraftName(product.name);
                setDraftPrice(product.price);
                setEditingId(product.id);
              }}
            >
              ✎
            </Button>
            <span className="w-12 text-right text-muted-foreground">
              ${product.price} ×
            </span>
            <input
              type="number"
              min={0}
              value={product.qty}
              onChange={(e) => setQty(product.id, e.target.valueAsNumber || 0)}
              className="w-14 rounded border bg-transparent px-1 py-0.5 text-right"
            />
          </div>
        ))}
        <div className="font-semibold">
          ${products.reduce((sum, p) => sum + p.price * p.qty, 0)} total
        </div>
        {editingId !== null && (
          <div className="fixed inset-0 z-50 grid place-items-center bg-black/40">
            <div className="grid w-64 gap-2 rounded-lg border bg-background p-4 shadow-lg">
              <div className="font-semibold">Edit product</div>
              <input
                value={draftName}
                onChange={(e) => setDraftName(e.target.value)}
                className="rounded border bg-transparent px-2 py-1 text-sm"
              />
              <div className="flex items-center gap-1 text-sm">
                <span className="text-muted-foreground">$</span>
                <input
                  type="number"
                  min={0}
                  value={draftPrice}
                  onChange={(e) => setDraftPrice(e.target.valueAsNumber || 0)}
                  className="flex-1 rounded border bg-transparent px-2 py-1"
                />
              </div>
              <div className="flex justify-end gap-2">
                <Button variant="outline" size="sm" onClick={() => setEditingId(null)}>
                  Cancel
                </Button>
                <Button size="sm" onClick={save}>
                  Save
                </Button>
              </div>
            </div>
          </div>
        )}
      </CardContent>
    </Card>
  );
}
